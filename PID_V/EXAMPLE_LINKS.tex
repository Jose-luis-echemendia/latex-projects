\section{Best Practices for Using Links}

\begin{enumeration}
    \item Use \code{\textbackslash link\{\}} when the URL itself is informative
    \begin{example}
    Project repository: \link{https://github.com/user/project-name}
    \end{example}

    \item Use \code{\textbackslash linktext\{\}\{\}} for cleaner, more readable text
    \begin{example}
    Read the \linktext{complete guide}{https://example.com/very/long/url/here}
    instead of showing a long URL.
    \end{example}

    \item Keep link text descriptive and meaningful
    \begin{bulletlist}
        \item \textcolor{green}{Good:} \linktext{Django REST framework tutorial}{https://www.django-rest-framework.org/tutorial/}
        \item \textcolor{red}{Bad:} \linktext{click here}{https://www.django-rest-framework.org/tutorial/}
    \end{bulletlist}

    \item Test all links before finalizing your document

    \item Consider using permanent URLs (DOIs) for academic papers
    \begin{example}
    \linktext{Paper title}{https://doi.org/10.1234/example}
    \end{example}
\end{enumeration}

\section{Quick Reference}

\subsection{Syntax}

\begin{codeblock}
% Visible URL link
\textbackslash link\{https://www.example.com/\}

% Custom text link
\textbackslash linktext\{descriptive text\}\{https://www.example.com/\}
\end{codeblock}

\subsection{Common Use Cases}

\begin{bulletlist}
    \item Repository links: \link{https://github.com/username/repo}
    \item Documentation: \linktext{Read the docs}{https://readthedocs.org/}
    \item Video tutorials: \linktext{YouTube channel}{https://www.youtube.com/channel/example}
    \item Academic papers: \linktext{DOI link}{https://doi.org/10.1234/example}
    \item API references: \linktext{API docs}{https://api.example.com/docs}
\end{bulletlist}
